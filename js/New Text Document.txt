  function fmt(n) {
    return (Number(n) || 0).toLocaleString(undefined, {
      maximumFractionDigits: 0
    });

  function fmt1(n) {
    return (Number(n) || 0).toLocaleString(undefined, {

  function pct(current, target) {
    const c = Number(current) || 0;
    const t = Number(target) || 0;
    return t ? (c / t) * 100 : 0;
  }

  function uniqueValues(records, field) {
    return [...new Set(records.map((r) => r[field]).filter(Boolean))].sort();
  }

  function sum(records, field) {
    return records.reduce((acc, r) => acc + (Number(r[field]) || 0), 0);
  }

  function setWarning(message) {
    const el = $("data-warning");
    if (!el) return;

    if (!message) {
      el.style.display = "none";
      el.textContent = "";
      return;
    }

    el.style.display = "block";
    el.textContent = message;
  }

  function shortLabel(text, max = 42) {
    const value = String(text || "");
    if (value.length <= max) return value;
    return value.slice(0, max - 3) + "...";
  }

  function populateSelect(id, values, label) {
    const el = $(id);
    if (!el) return;

    const current = el.value;
    el.innerHTML = "";

    const all = document.createElement("option");
    all.value = "All";
    all.textContent = `All ${label}`;
    el.appendChild(all);

    values.forEach((v) => {
      const opt = document.createElement("option");
      opt.value = v;
      opt.textContent = v;
      el.appendChild(opt);
    });

    if ([...el.options].some((o) => o.value === current)) {
      el.value = current;
    }
  }

  function onFilterChange() {
    const state = $("state-filter")?.value || "All";
    const countyEl = $("county-filter");

    if (countyEl && this.id === "state-filter") {
      const counties =
        state === "All"
          ? uniqueValues(DATA, "county")
          : uniqueValues(DATA.filter((r) => r.state === state), "county");

      populateSelect("county-filter", counties, "Counties");
    }

    updateDashboard();
  }

  function getFilteredRecords() {
    const indicator = $("indicator-filter")?.value || "All";
    const agency = $("agency-filter")?.value || "All";
    const state = $("state-filter")?.value || "All";
    const county = $("county-filter")?.value || "All";

    return DATA.filter((r) => {
      return (
        (indicator === "All" || r.indicator === indicator) &&
        (agency === "All" || r.agency === agency) &&
        (state === "All" || r.state === state) &&
        (county === "All" || r.county === county)
      );
    });
  }

  function groupSum(records, key, valueField) {
    const out = {};

    records.forEach((r) => {
      const k = r[key] || "Not specified";
      out[k] = (out[k] || 0) + (Number(r[valueField]) || 0);
    });

    return Object.entries(out)
      .map(([name, value]) => ({ name, value }))
      .sort((a, b) => b.value - a.value);
  }

  function groupIndicators(records) {
    const out = {};

    records.forEach((r) => {
      const indicator = r.indicator || "Not specified";

      if (!out[indicator]) {
        out[indicator] = {
          indicator,
          agencies: new Set(),
          current: 0,
          target: Number(r.target) || 0
        };
      }

      out[indicator].current += Number(r.current) || 0;

      if (r.agency) out[indicator].agencies.add(r.agency);

      if (!out[indicator].target && Number(r.target)) {
        out[indicator].target = Number(r.target);
      }
    });

    return Object.values(out)
      .map((d) => ({
        indicator: d.indicator,
        agencies: [...d.agencies].sort().join(", "),
        current: d.current,
        target: d.target,
        achieved: pct(d.current, d.target)
      }))
      .sort((a, b) => b.current - a.current);
  }

  function plotBar(id, rows, orientation = "h", color = CHART_BLUE, xTitle = "Current Number") {
    const el = $(id);
    if (!el || typeof Plotly === "undefined") return;

    if (!rows.length) {
      Plotly.newPlot(
        id,
        [],
        darkPlotLayout({
          title: {
            text: "No data",
            font: { color: "#e8f1fa", size: 13 }
          },
          margin: { t: 30, r: 20, b: 40, l: 120 }
        }),
        { displayModeBar: false, responsive: true }
      );
      return;
    }

    const chartRows = orientation === "h" ? [...rows].reverse() : rows;

    const trace =
      orientation === "h"
        ? {
            type: "bar",
            orientation: "h",
            y: chartRows.map((d) => shortLabel(d.name, 36)),
            x: chartRows.map((d) => d.value),
            text: chartRows.map((d) => fmt(d.value)),
            textposition: "auto",
            marker: { color },
            customdata: chartRows.map((d) => d.name),
            hovertemplate: "%{customdata}<br>%{x:,}<extra></extra>"
          }
        : {
            type: "bar",
            x: chartRows.map((d) => shortLabel(d.name, 22)),
            y: chartRows.map((d) => d.value),
            text: chartRows.map((d) => fmt(d.value)),
            textposition: "auto",
            marker: { color },
            customdata: chartRows.map((d) => d.name),
            hovertemplate: "%{customdata}<br>%{y:,}<extra></extra>"
          };

    Plotly.newPlot(
      id,
      [trace],
      darkPlotLayout({
        margin: {
          t: 25,
          r: 25,
          b: orientation === "h" ? 40 : 90,
          l: orientation === "h" ? 180 : 55
        },
        xaxis: {
          title: orientation === "h" ? xTitle : "",
          gridcolor: "rgba(255,255,255,0.08)",
          zerolinecolor: "rgba(255,255,255,0.15)",
          color: "#8ba8c4"
        },
        yaxis: {
          title: "",
          gridcolor: "rgba(255,255,255,0.05)",
          zerolinecolor: "rgba(255,255,255,0.15)",
          color: "#8ba8c4",
          automargin: true
        }
      }),
      { displayModeBar: false, responsive: true }
    );
  }

  function updateKPIs(records) {
    const states = uniqueValues(records.filter((r) => !r.isAdminArea), "state").length;
    const adminAreas = uniqueValues(records.filter((r) => r.isAdminArea), "state").length;

    if ($("kpi-states")) $("kpi-states").textContent = fmt(states);
    if ($("kpi-admin-areas")) $("kpi-admin-areas").textContent = fmt(adminAreas);
    if ($("kpi-counties")) $("kpi-counties").textContent = fmt(uniqueValues(records, "county").length);
    if ($("kpi-agencies")) $("kpi-agencies").textContent = fmt(uniqueValues(records, "agency").length);
    if ($("kpi-indicators")) $("kpi-indicators").textContent = fmt(uniqueValues(records, "indicator").length);
    if ($("kpi-male")) $("kpi-male").textContent = fmt(sum(records, "male"));
    if ($("kpi-female")) $("kpi-female").textContent = fmt(sum(records, "female"));
  }

  function updateInsights(records) {
    const el = $("simple-insights-list");
    if (!el) return;

    if (!records.length) {
      el.innerHTML = `<div class="simple-insight-item">No records match the selected filters.</div>`;
      return;
    }

    const indicatorRows = groupIndicators(records);
    const topIndicator = indicatorRows[0];
    const topState = groupSum(records, "state", "current")[0];
    const topCounty = groupSum(records, "county", "current")[0];

    el.innerHTML = `
      <div class="simple-insight-item"><strong>${fmt(uniqueValues(records, "indicator").length)}</strong> indicators are reported across <strong>${fmt(uniqueValues(records, "county").length)}</strong> counties.</div>
      <div class="simple-insight-item">Highest current number is for <strong>${shortLabel(topIndicator?.indicator || "N/A", 70)}</strong>.</div>
      <div class="simple-insight-item">Top state/admin area is <strong>${topState?.name || "N/A"}</strong>, with <strong>${fmt(topState?.value || 0)}</strong>.</div>
      <div class="simple-insight-item">Top county is <strong>${topCounty?.name || "N/A"}</strong>, with <strong>${fmt(topCounty?.value || 0)}</strong>.</div>
      <div class="simple-insight-item">Reported gender breakdown: <strong>${fmt(sum(records, "male"))}</strong> male and <strong>${fmt(sum(records, "female"))}</strong> female.</div>
    `;
  }

    lastTables.indicatorTable = rows;

        </tr>
      `

        .join("") || `<tr><td colspan="5">No data</td></tr>`;
  }

  function initMap() {
    if (map || !$("peace-map")) return;

    map = L.map("peace-map", { scrollWheelZoom: false }).setView(
      SOUTH_SUDAN_CENTER,
      6
    );

    L.tileLayer("https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png", {
      maxZoom: 18,
      attribution: "&copy; OpenStreetMap contributors"
    }).addTo(map);

    markersLayer = L.layerGroup().addTo(map);
  }

  function updateMap(records) {
    initMap();
    if (!map || !markersLayer) return;

    markersLayer.clearLayers();

    const byCounty = {};

    records.forEach((r) => {
      const key = r.county;

      if (!byCounty[key]) {
        byCounty[key] = {
          county: r.county,
          state: r.state,
          indicators: new Set(),
          agencies: new Set(),
          current: 0
        };
      }

      if (r.indicator) byCounty[key].indicators.add(r.indicator);
      if (r.agency) byCounty[key].agencies.add(r.agency);
      byCounty[key].current += Number(r.current) || 0;
    });

    const points = Object.values(byCounty).filter((d) => COUNTY_COORDS[d.county]);
    const maxIndicators = Math.max(...points.map((d) => d.indicators.size), 1);
    const bounds = [];

    points.forEach((d) => {
      const coords = COUNTY_COORDS[d.county];
      const count = d.indicators.size;
      const radius = 7 + Math.sqrt(count / maxIndicators) * 24;

      const marker = L.circleMarker(coords, {
        radius,
        color: CHART_GREEN,
        fillColor: CHART_GREEN,
        fillOpacity: 0.65,
        weight: 1
      }).bindPopup(`
        <strong>${d.county}</strong><br/>
        State/Admin Area: ${d.state}<br/>
        Indicators: ${fmt(count)}<br/>
        Current Number: ${fmt(d.current)}<br/>
        Reporting Entities: ${[...d.agencies].join(", ")}
      `);

      marker.addTo(markersLayer);
      bounds.push(coords);
    });

    if (bounds.length) {
      map.fitBounds(bounds, { padding: [40, 40], maxZoom: 7 });
    } else {
      map.setView(SOUTH_SUDAN_CENTER, 6);
    }

    if ($("map-counties")) $("map-counties").textContent = fmt(uniqueValues(records, "county").length);
    if ($("map-indicators")) $("map-indicators").textContent = fmt(uniqueValues(records, "indicator").length);
    if ($("map-current")) $("map-current").textContent = fmt(sum(records, "current"));
    if ($("map-agencies")) $("map-agencies").textContent = fmt(uniqueValues(records, "agency").length);
  }

  function renderCharts(records) {
    const indicatorRows = groupIndicators(records);

    lastTables.indicatorAchievement = indicatorRows.map((r) => ({
      name: r.indicator,
      value: r.achieved,
      current: r.current,
      target: r.target
    }));

    const achievementRows = lastTables.indicatorAchievement.sort((a, b) => b.value - a.value);

    plotBar(
      "indicator-achievement-chart",
      achievementRows,
      "h",
      CHART_GREEN,
      "% Achieved"
    );

    const stateRows = groupSum(records, "state", "current").slice(0, 12);
    const countyRows = groupSum(records, "county", "current").slice(0, 12);
    const agencyRows = groupSum(records, "agency", "current").slice(0, 12);

    lastTables.state = stateRows;
    lastTables.county = countyRows;
    lastTables.agency = agencyRows;
    lastTables.gender = [
      { name: "Male", value: sum(records, "male") },
      { name: "Female", value: sum(records, "female") }
    ];

    plotBar("state-chart", stateRows, "h", CHART_BLUE, "Current Number");
    plotBar("county-chart", countyRows, "h", CHART_ORANGE, "Current Number");
    plotBar("agency-chart", agencyRows, "h", CHART_BLUE, "Current Number");

    if ($("gender-chart") && typeof Plotly !== "undefined") {
      Plotly.newPlot(
        "gender-chart",
        [
          {
            type: "pie",
            labels: ["Male", "Female"],
            values: [sum(records, "male"), sum(records, "female")],
            hole: 0.35,
            textinfo: "label+percent",
            marker: {
              colors: [MALE_COLOR, FEMALE_COLOR]
            },
            hovertemplate: "%{label}<br>%{value:,}<br>%{percent}<extra></extra>"
          }
        ],
        darkPlotLayout({
          margin: { t: 20, r: 20, b: 20, l: 20 },
          showlegend: true,
          legend: {
            font: { color: "#e8f1fa" }
          }
        }),
        { displayModeBar: false, responsive: true }
      );
    }

    renderSelectedIndicatorCharts(records);
  }

  function renderSelectedIndicatorCharts(records) {
    const selectedIndicator = $("indicator-filter")?.value || "All";

    let selectedRecords = records;

    if (selectedIndicator === "All") {
      const topIndicator = groupIndicators(records)[0]?.indicator;
      selectedRecords = topIndicator
        ? records.filter((r) => r.indicator === topIndicator)
        : records;
    }

    const stateRows = groupSum(selectedRecords, "state", "current").slice(0, 5);

    plotBar(
      "selected-indicator-state-chart",
      stateRows,
      "h",
      CHART_GREEN,
      "Current Number"
    );

    if ($("selected-indicator-gender-chart") && typeof Plotly !== "undefined") {
      Plotly.newPlot(
        "selected-indicator-gender-chart",
        [
          {
            type: "bar",
            x: stateRows.map((d) => shortLabel(d.name, 20)),
            y: stateRows.map((d) => {
              const stateRecords = selectedRecords.filter((r) => r.state === d.name);
              return sum(stateRecords, "male");
            }),
            name: "Male",
            marker: { color: MALE_COLOR }
          },
          {
            type: "bar",
            x: stateRows.map((d) => shortLabel(d.name, 20)),
            y: stateRows.map((d) => {
              const stateRecords = selectedRecords.filter((r) => r.state === d.name);
              return sum(stateRecords, "female");
            }),
            name: "Female",
            marker: { color: FEMALE_COLOR }
          }
        ],
        darkPlotLayout({
          barmode: "stack",
          margin: { t: 25, r: 25, b: 90, l: 55 },
          showlegend: true,
          legend: {
            font: { color: "#e8f1fa" }
          }
        }),
        { displayModeBar: false, responsive: true }
      );
    }
  }

  function renderRemarks(records) {
    const tbody = $("remarks-table");
    if (!tbody) return;

    const rows = records
      .filter((r) => r.remarks || r.risk || r.implementingPartners)
      .slice(0, 80);

    lastTables.remarks = rows;

    tbody.innerHTML =
      rows
        .map((r) => {
          const combined = [
            r.remarks ? `Remarks: ${r.remarks}` : "",
            r.risk ? `Risk: ${r.risk}` : "",
            r.implementingPartners ? `Partners: ${r.implementingPartners}` : ""
          ].filter(Boolean).join("<br/>");

          return `
            <tr>
              <td>${r.state || ""}</td>
              <td>${r.county || ""}</td>
              <td>${r.indicator || ""}</td>
              <td>${r.agency || ""}</td>
              <td>${fmt(r.current)}</td>
              <td>${combined}</td>
            </tr>
          `;
        })
        .join("") || `<tr><td colspan="6">No remarks, risks or partners reported for selected filters.</td></tr>`;
  }

  function updateDashboard() {
    const records = getFilteredRecords();

    updateKPIs(records);
    updateInsights(records);
    renderIndicatorTable(records);
    updateMap(records);
    renderCharts(records);
    renderRemarks(records);
  }

  window.downloadChartPNG = function (chartId, filename) {
    if (typeof Plotly === "undefined") return;

    Plotly.downloadImage(chartId, {
      format: "png",
      filename: filename || chartId,
      height: 650,
      width: 1000,
      scale: 2
    });
  };

  window.downloadChartCSV = function (type) {
    const rows = lastTables[type] || [];
    if (!rows.length) return;

    let csv = "";

    if (type === "indicatorTable") {
      csv =
        "Indicator,Reporting UN Entities,Current Number,Target,Achieved %\n" +
        rows
          .map(
            (r) =>
              `"${r.indicator.replace(/"/g, '""')}","${r.agencies.replace(/"/g, '""')}",${r.current},${r.target},${r.achieved}`
          )
          .join("\n");
    } else if (type === "indicatorAchievement") {
      csv =
        "Indicator,Achieved %,Current Number,Target\n" +
        rows
          .map(
            (r) =>
              `"${r.name.replace(/"/g, '""')}",${r.value},${r.current},${r.target}`
          )
          .join("\n");
    } else if (type === "remarks") {
      csv =
        "State,County,Indicator,Agency,Current,Remarks,Risk,Implementing Partners\n" +
        rows
          .map(
            (r) =>
              `"${r.state}","${r.county}","${(r.indicator || "").replace(/"/g, '""')}","${r.agency}",${r.current},"${(r.remarks || "").replace(/"/g, '""')}","${(r.risk || "").replace(/"/g, '""')}","${(r.implementingPartners || "").replace(/"/g, '""')}"`
          )
          .join("\n");
    } else {
      csv =
        "Name,Value\n" +
        rows.map((r) => `"${r.name.replace(/"/g, '""')}",${r.value}`).join("\n");
    }

    const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
    const url = URL.createObjectURL(blob);

    const a = document.createElement("a");
    a.href = url;
    a.download = `peace_governance_${type}.csv`;

    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);

    URL.revokeObjectURL(url);
  };

  document.addEventListener("DOMContentLoaded", () => {
    if (!DATA.length) {
      setWarning(
        "No Peace and Governance records found. Please run generate-data-js.py and confirm the All Outputs sheet has Output = Peace and Governance."
      );
    }

    initFilters();
    initMap();
    updateDashboard();
  });
